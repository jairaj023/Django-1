loop control structure
	-> is used to repeat some set of statements for repeatitive time
	
	three types of loop

	1. while
	2. for
-----------------------------------------------------------------------
i=0     #starting

while i<=10:    #condition
    print(i)
    i=i+1       #incr.
------------------------------------------------------------------------
for i in range(1,11):
    print(i)

break
	-> is used to take cursor to the end of the loop
--------------------------------------------------------------------------
for i in range(1,20):
    if i>10:
        break
    else:
        print(i,end=", ")
--------------------------------------------------------------------------
continue
	-> is used to take cursor to the beginning of the loop
-------------------------------------------------------------------------
for i in range(1,20):
    if i<=10:
        continue
    else:
        print(i,end=", ")

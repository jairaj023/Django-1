----------------------------------------------------------------------------
< (less than) operator
----------------------------------------------------------------------------
i=8

if i<10:
    print("won")

else:
    print("loose")

-----------------------------------------------------------------------------
<= (less than or equals) operator
-----------------------------------------------------------------------------
i=8

if i<=8:
    print("won")

else:
    print("loose")



----------------------------------------------------------------------------
> (greater than) operator
----------------------------------------------------------------------------
i=8

if i>10:
    print("won")

else:
    print("loose")



---------------------------------------------------------------------------
== (comparision) operator
	-> equals
--------------------------------------------------------------------------
i=8

if i==8:
    print("won")

else:
    print("loose")



---------------------------------------------------------------------------
!= (not equals) operator
---------------------------------------------------------------------------
i=8

if i!=10:
    print("won")

else:
    print("loose")
